cmake_minimum_required(VERSION 3.25)

list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
if(NOT CMAKE_TOOLCHAIN_FILE OR NOT EXISTS "${CMAKE_TOOLCHAIN_FILE}")
    message(FATAL_ERROR "Set VCPKG_ROOT to an existing vcpkg checkout and configure with a Stone CMake preset.")
endif()
project(Stone LANGUAGES C CXX RC)

file(GLOB_RECURSE stone_vcpkg_configuration CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/dep/vcpkg-overlay-ports/*"
    "${CMAKE_CURRENT_SOURCE_DIR}/dep/vcpkg-overlay-triplets/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/vcpkg.json" ${stone_vcpkg_configuration})

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Stone requires a Windows target and an MSVC-compatible toolchain.")
endif()

string(TOLOWER "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}" STONE_WINDOWS_ARCH)
if(STONE_WINDOWS_ARCH STREQUAL "x64")
    enable_language(ASM_MASM)
elseif(STONE_WINDOWS_ARCH STREQUAL "arm64")
    set(STONE_WINDOWS_ARCH "ARM64")
else()
    message(FATAL_ERROR "Stone supports Windows x64 and ARM64. Detected: ${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}")
endif()

get_property(stone_multi_config GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
if(NOT stone_multi_config AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build configuration" FORCE)
endif()

set(stone_platform_bin "${PROJECT_SOURCE_DIR}/bin/${STONE_WINDOWS_ARCH}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${stone_platform_bin}/$<CONFIG>")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/$<CONFIG>")

include(StoneVersion)
configure_file(cmake/StoneVersion.h.in "${CMAKE_BINARY_DIR}/version/StoneVersion.h" @ONLY)
include(WindowsDependencies)

option(STONE_TESTS "Build Windows x64 component tests and Android fixture targets" OFF)
option(STONE_BUILD_SETTINGS "Build the WinUI settings application (.NET 10 SDK required)" ON)
option(STONE_PACKAGING "Enable MSI and MSIX packaging targets" OFF)
if(STONE_TESTS AND STONE_WINDOWS_ARCH STREQUAL "x64")
    enable_testing()
endif()

add_subdirectory(src/windows/image-tools)
add_subdirectory(src/android/runtime)
add_subdirectory(src/windows/ipc)
add_subdirectory(src/windows/host)
add_subdirectory(src/windows/service)
if(STONE_WINDOWS_ARCH STREQUAL "x64")
    add_subdirectory(src/windows/subsystem)
endif()
add_subdirectory(src/windows/client)
if(TARGET StoneArtCompiler)
    add_dependencies(StoneClient StoneArtCompiler)
endif()
add_dependencies(StoneService StoneClient)
foreach(version_target StoneClient StoneService StoneImageTool StoneArtCompiler)
    if(TARGET ${version_target})
        target_include_directories(${version_target} PRIVATE "$<$<COMPILE_LANGUAGE:RC>:${CMAKE_BINARY_DIR}/version>")
        if(NOT version_target MATCHES "^(StoneClient|StoneService)$")
            configure_file(cmake/StoneVersion.rc.in "${CMAKE_BINARY_DIR}/version/${version_target}.rc" @ONLY)
            target_sources(${version_target} PRIVATE "${CMAKE_BINARY_DIR}/version/${version_target}.rc")
        endif()
    endif()
endforeach()
if(STONE_BUILD_SETTINGS)
    add_subdirectory(src/windows/settings)
endif()

if(STONE_TESTS AND STONE_WINDOWS_ARCH STREQUAL "x64")
    add_subdirectory(test/windows)
endif()

if(STONE_PACKAGING)
    include(WindowsPackaging)
    add_subdirectory(msipackage)
    add_subdirectory(msixgluepackage)
    add_subdirectory(msixinstaller)
endif()
